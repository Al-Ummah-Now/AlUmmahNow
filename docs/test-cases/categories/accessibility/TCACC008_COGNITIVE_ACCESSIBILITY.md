# TCACC008: Cognitive Accessibility

## Test Case Information

- **Test Case ID**: TCACC008
- **Category**: Accessibility
- **Sub-category**: Cognitive Accessibility
- **Priority**: High
- **Estimated Time**: 90 minutes
- **Target Browsers**: Chrome, Firefox, Safari, Edge
- **Target Devices**: Desktop, Tablet, Mobile
- **Accessibility Standards**: WCAG 2.1 Level AA, Cognitive Accessibility Guidelines

## Test Description

This test case validates the cognitive accessibility of the Al Ummah Now platform, ensuring that users with cognitive disabilities, learning disabilities, and neurodiverse conditions can effectively use all features. It covers readability, comprehension, navigation simplicity, error prevention, and cognitive load management.

## Test Objectives

1. Verify content readability and comprehension
2. Test navigation simplicity and predictability
3. Validate error prevention and recovery mechanisms
4. Ensure consistent design patterns and interactions
5. Test help and support features accessibility
6. Verify attention and focus management
7. Test memory aids and cognitive support features
8. Validate multi-modal information presentation

## Prerequisites

- Users with cognitive disabilities for testing (if available)
- Cognitive accessibility testing tools
- Readability analysis tools
- Attention and focus testing methodologies
- Various user personas with different cognitive needs
- Assistive technology for cognitive support

### Cognitive Accessibility JavaScript
```javascript
class CognitiveAccessibility {
  constructor() {
    this.setupCognitiveSupport();
    this.setupReadabilityEnhancements();
    this.setupNavigationAids();
    this.setupErrorPrevention();
    this.setupAttentionManagement();
  }

  setupCognitiveSupport() {
    // Detect user preferences for cognitive support
    this.detectCognitiveNeeds();
    
    // Setup cognitive support features
    this.setupMemoryAids();
    this.setupLearningSupport();
    this.setupPacingControl();
  }

  setupReadabilityEnhancements() {
    // Analyze and enhance text readability
    this.enhanceTextReadability();
    
    // Provide reading aids
    this.setupReadingAids();
    
    // Setup vocabulary support
    this.setupVocabularySupport();
  }

  setupNavigationAids() {
    // Simplify navigation
    this.simplifyNavigation();
    
    // Add wayfinding support
    this.addWayfindingAids();
    
    // Provide navigation guidance
    this.setupNavigationGuidance();
  }

  setupErrorPrevention() {
    // Prevent common errors
    this.preventCommonErrors();
    
    // Enhance error messages
    this.enhanceErrorMessages();
    
    // Provide recovery support
    this.setupErrorRecovery();
  }

  setupAttentionManagement() {
    // Minimize distractions
    this.minimizeDistractions();
    
    // Support focus
    this.setupFocusSupport();
    
    // Manage cognitive load
    this.manageCognitiveLoad();
  }

  detectCognitiveNeeds() {
    // Check for accessibility preferences
    const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
    const prefersHighContrast = window.matchMedia('(prefers-contrast: high)').matches;
    
    // Check for cognitive support preferences
    const cognitiveSupport = localStorage.getItem('cognitive-support');
    
    this.userPreferences = {
      reducedMotion: prefersReducedMotion,
      highContrast: prefersHighContrast,
      cognitiveSupport: cognitiveSupport === 'enabled',
      simplifiedMode: cognitiveSupport === 'simplified'
    };
  }

  enhanceTextReadability() {
    // Analyze text content
    const textElements = document.querySelectorAll('p, li, dd, .content-text');
    
    textElements.forEach(element => {
      const text = element.textContent;
      const readability = this.analyzeReadability(text);
      
      // Enhance if readability is poor
      if (readability.fleschKincaidGrade > 8) {
        this.simplifyText(element);
      }
      
      // Add reading aids if needed
      if (this.userPreferences.cognitiveSupport) {
        this.addReadingAids(element);
      }
    });
  }

  analyzeReadability(text) {
    // Simple readability analysis
    const sentences = text.split(/[.!?]+/).filter(s => s.trim().length > 0);
    const words = text.split(/\s+/).filter(w => w.length > 0);
    const syllables = words.reduce((total, word) => total + this.countSyllables(word), 0);
    const complexWords = words.filter(word => this.countSyllables(word) > 2).length;
    
    const averageWordsPerSentence = words.length / sentences.length;
    const averageSyllablesPerWord = syllables / words.length;
    
    // Flesch-Kincaid Grade Level
    const fleschKincaidGrade = (0.39 * averageWordsPerSentence) + (11.8 * averageSyllablesPerWord) - 15.59;
    
    // Flesch Reading Ease
    const fleschReadingEase = 206.835 - (1.015 * averageWordsPerSentence) - (84.6 * averageSyllablesPerWord);
    
    return {
      fleschKincaidGrade,
      fleschReadingEase,
      averageWordsPerSentence,
      averageSyllablesPerWord,
      percentageComplexWords: (complexWords / words.length) * 100
    };
  }

  countSyllables(word) {
    word = word.toLowerCase();
    if (word.length <= 3) return 1;
    
    word = word.replace(/(?:[^laeiouy]es|ed|[^laeiouy]e)$/, '');
    word = word.replace(/^y/, '');
    const matches = word.match(/[aeiouy]{1,2}/g);
    return matches ? matches.length : 1;
  }

  simplifyText(element) {
    const text = element.textContent;
    
    // Replace complex words with simpler alternatives
    const simplifications = {
      'utilize': 'use',
      'accomplish': 'do',
      'sufficient': 'enough',
      'demonstrate': 'show',
      'approximately': 'about',
      'subsequently': 'then',
      'consequently': 'so',
      'nevertheless': 'still',
      'furthermore': 'also',
      'therefore': 'so'
    };
    
    let simplifiedText = text;
    Object.entries(simplifications).forEach(([complex, simple]) => {
      const regex = new RegExp(`\\b${complex}\\b`, 'gi');
      simplifiedText = simplifiedText.replace(regex, simple);
    });
    
    // Break up long sentences
    const sentences = simplifiedText.split(/[.!?]+/);
    const simplifiedSentences = sentences.map(sentence => {
      const words = sentence.trim().split(/\s+/);
      if (words.length > 15) {
        // Break long sentences into smaller ones
        const midpoint = Math.floor(words.length / 2);
        return [
          words.slice(0, midpoint).join(' '),
          words.slice(midpoint).join(' ')
        ].join('. ');
      }
      return sentence;
    });
    
    element.textContent = simplifiedSentences.join('. ');
  }

  addReadingAids(element) {
    // Add reading controls
    const readingControls = document.createElement('div');
    readingControls.className = 'reading-controls';
    readingControls.innerHTML = `
      <button class="reading-font-size" aria-label="Increase font size">A+</button>
      <button class="reading-spacing" aria-label="Increase line spacing">↕</button>
      <button class="reading-highlight" aria-label="Highlight text">🖍</button>
      <button class="reading-speak" aria-label="Read text aloud">🔊</button>
    `;
    
    element.parentNode.insertBefore(readingControls, element);
    
    // Add event listeners
    this.setupReadingControls(readingControls, element);
  }

  setupReadingControls(controls, textElement) {
    const fontSizeBtn = controls.querySelector('.reading-font-size');
    const spacingBtn = controls.querySelector('.reading-spacing');
    const highlightBtn = controls.querySelector('.reading-highlight');
    const speakBtn = controls.querySelector('.reading-speak');
    
    let currentFontSize = 16;
    let currentSpacing = 1.5;
    let isHighlighted = false;
    
    fontSizeBtn.addEventListener('click', () => {
      currentFontSize = currentFontSize >= 24 ? 16 : currentFontSize + 2;
      textElement.style.fontSize = `${currentFontSize}px`;
      this.announce(`Font size: ${currentFontSize} pixels`);
    });
    
    spacingBtn.addEventListener('click', () => {
      currentSpacing = currentSpacing >= 2.5 ? 1.5 : currentSpacing + 0.25;
      textElement.style.lineHeight = currentSpacing;
      this.announce(`Line spacing: ${currentSpacing}`);
    });
    
    highlightBtn.addEventListener('click', () => {
      isHighlighted = !isHighlighted;
      if (isHighlighted) {
        textElement.style.backgroundColor = '#ffff99';
        textElement.style.padding = '4px';
        this.announce('Text highlighted');
      } else {
        textElement.style.backgroundColor = '';
        textElement.style.padding = '';
        this.announce('Highlight removed');
      }
    });
    
    speakBtn.addEventListener('click', () => {
      if ('speechSynthesis' in window) {
        const utterance = new SpeechSynthesisUtterance(textElement.textContent);
        utterance.rate = 0.8;
        speechSynthesis.speak(utterance);
        this.announce('Reading text aloud');
      }
    });
  }

  setupVocabularySupport() {
    // Find technical terms and provide definitions
    const technicalTerms = ['prayer', 'zakat', 'hajj', 'ramadan', 'quran', 'sunnah'];
    
    technicalTerms.forEach(term => {
      const regex = new RegExp(`\\b${term}\\b`, 'gi');
      document.body.innerHTML = document.body.innerHTML.replace(regex, 
        `<span class="technical-term" data-term="${term}" tabindex="0" role="button" aria-label="Definition of ${term}">${term}</span>`
      );
    });
    
    // Add click handlers for definitions
    document.querySelectorAll('.technical-term').forEach(element => {
      element.addEventListener('click', (e) => this.showDefinition(e.target));
      element.addEventListener('keypress', (e) => {
        if (e.key === 'Enter' || e.key === ' ') {
          e.preventDefault();
          this.showDefinition(e.target);
        }
      });
    });
  }

  showDefinition(element) {
    const term = element.dataset.term;
    const definitions = {
      'prayer': 'The act of communicating with Allah through prescribed words and actions',
      'zakat': 'The obligatory charity given to those in need, one of the Five Pillars of Islam',
      'hajj': 'The annual pilgrimage to Mecca, required of all able Muslims once in their lifetime',
      'ramadan': 'The ninth month of the Islamic calendar, a time of fasting and spiritual reflection',
      'quran': 'The holy book of Islam, believed to be the word of Allah as revealed to Prophet Muhammad',
      'sunnah': 'The practices and traditions of Prophet Muhammad, serving as a model for Muslims'
    };
    
    const definition = definitions[term.toLowerCase()];
    if (definition) {
      this.showTooltip(element, definition);
      this.announce(`Definition of ${term}: ${definition}`);
    }
  }

  showTooltip(element, text) {
    // Remove existing tooltips
    const existingTooltip = document.querySelector('.definition-tooltip');
    if (existingTooltip) {
      existingTooltip.remove();
    }
    
    // Create new tooltip
    const tooltip = document.createElement('div');
    tooltip.className = 'definition-tooltip';
    tooltip.textContent = text;
    tooltip.setAttribute('role', 'tooltip');
    tooltip.setAttribute('aria-live', 'polite');
    
    // Position tooltip
    const rect = element.getBoundingClientRect();
    tooltip.style.position = 'absolute';
    tooltip.style.top = `${rect.bottom + window.scrollY + 5}px`;
    tooltip.style.left = `${rect.left + window.scrollX}px`;
    tooltip.style.maxWidth = '300px';
    tooltip.style.padding = '8px';
    tooltip.style.backgroundColor = '#f0f0f0';
    tooltip.style.border = '1px solid #ccc';
    tooltip.style.borderRadius = '4px';
    tooltip.style.fontSize = '14px';
    tooltip.style.zIndex = '1000';
    
    document.body.appendChild(tooltip);
    
    // Remove tooltip on click outside
    setTimeout(() => {
      document.addEventListener('click', function removeTooltip(e) {
        if (!tooltip.contains(e.target)) {
          tooltip.remove();
          document.removeEventListener('click', removeTooltip);
        }
      });
    }, 100);
  }

  simplifyNavigation() {
    // Simplify main navigation
    const mainNav = document.querySelector('nav, [role="navigation"]');
    if (mainNav) {
      // Limit menu items
      const navItems = mainNav.querySelectorAll('li, [role="menuitem"]');
      if (navItems.length > 7) {
        // Group less important items
        const moreItem = document.createElement('li');
        moreItem.innerHTML = `
          <button aria-expanded="false" aria-haspopup="true">More</button>
          <ul role="menu" hidden></ul>
        `;
        
        const extraItems = Array.from(navItems).slice(7);
        const submenu = moreItem.querySelector('[role="menu"]');
        extraItems.forEach(item => submenu.appendChild(item));
        
        mainNav.appendChild(moreItem);
      }
    }
    
    // Add breadcrumbs if not present
    if (!document.querySelector('.breadcrumbs')) {
      this.addBreadcrumbs();
    }
  }

  addBreadcrumbs() {
    const path = window.location.pathname;
    const pathSegments = path.split('/').filter(segment => segment);
    
    if (pathSegments.length > 1) {
      const breadcrumbs = document.createElement('nav');
      breadcrumbs.setAttribute('aria-label', 'Breadcrumb navigation');
      breadcrumbs.className = 'breadcrumbs';
      
      const breadcrumbList = document.createElement('ol');
      breadcrumbList.setAttribute('role', 'list');
      
      // Add home
      const homeItem = document.createElement('li');
      homeItem.innerHTML = '<a href="/">Home</a>';
      breadcrumbList.appendChild(homeItem);
      
      // Add path segments
      let currentPath = '';
      pathSegments.forEach((segment, index) => {
        currentPath += '/' + segment;
        const isLast = index === pathSegments.length - 1;
        
        const item = document.createElement('li');
        if (isLast) {
          item.innerHTML = `<span aria-current="page">${this.formatBreadcrumbText(segment)}</span>`;
        } else {
          item.innerHTML = `<a href="${currentPath}">${this.formatBreadcrumbText(segment)}</a>`;
        }
        breadcrumbList.appendChild(item);
      });
      
      breadcrumbs.appendChild(breadcrumbList);
      
      // Insert at top of main content
      const main = document.querySelector('main, [role="main"]');
      if (main) {
        main.insertBefore(breadcrumbs, main.firstChild);
      }
    }
  }

  formatBreadcrumbText(segment) {
    return segment
      .replace(/-/g, ' ')
      .replace(/\b\w/g, l => l.toUpperCase());
  }

  addWayfindingAids() {
    // Add page title enhancement
    const pageTitle = document.querySelector('h1');
    if (pageTitle) {
      const wayfindingInfo = document.createElement('div');
      wayfindingInfo.className = 'wayfinding-info';
      wayfindingInfo.setAttribute('aria-live', 'polite');
      
      const section = this.getCurrentSection();
      const subsection = this.getCurrentSubsection();
      
      let info = `You are in: ${section}`;
      if (subsection) {
        info += `, ${subsection}`;
      }
      
      wayfindingInfo.textContent = info;
      pageTitle.parentNode.insertBefore(wayfindingInfo, pageTitle.nextSibling);
    }
    
    // Add progress indicators for multi-step processes
    this.addProgressIndicators();
  }

  getCurrentSection() {
    const path = window.location.pathname;
    if (path.includes('/videos')) return 'Videos';
    if (path.includes('/channels')) return 'Channels';
    if (path.includes('/profile')) return 'Profile';
    if (path.includes('/settings')) return 'Settings';
    return 'Home';
  }

  getCurrentSubsection() {
    const path = window.location.pathname;
    const segments = path.split('/').filter(s => s);
    if (segments.length > 2) {
      return this.formatBreadcrumbText(segments[segments.length - 1]);
    }
    return null;
  }

  addProgressIndicators() {
    // Find multi-step forms or processes
    const multiStepForms = document.querySelectorAll('.multi-step, .wizard, [data-steps]');
    
    multiStepForms.forEach(form => {
      const steps = form.querySelectorAll('.step, [data-step]');
      if (steps.length > 1) {
        const progressIndicator = document.createElement('div');
        progressIndicator.className = 'progress-indicator';
        progressIndicator.setAttribute('aria-label', 'Progress');
        
        const progressList = document.createElement('ol');
        progressList.setAttribute('role', 'list');
        
        steps.forEach((step, index) => {
          const isCompleted = step.classList.contains('completed');
          const isCurrent = step.classList.contains('current') || index === 0;
          
          const item = document.createElement('li');
          item.className = isCompleted ? 'completed' : isCurrent ? 'current' : 'upcoming';
          item.innerHTML = `
            <span class="step-number">${index + 1}</span>
            <span class="step-label">${step.querySelector('h2, h3, legend')?.textContent || `Step ${index + 1}`}</span>
          `;
          
          if (isCurrent) {
            item.setAttribute('aria-current', 'step');
          }
          
          progressList.appendChild(item);
        });
        
        progressIndicator.appendChild(progressList);
        form.insertBefore(progressIndicator, form.firstChild);
      }
    });
  }

  preventCommonErrors() {
    // Add confirmation for destructive actions
    const destructiveButtons = document.querySelectorAll('button[formaction*="delete"], button[onclick*="delete"], .btn-danger');
    
    destructiveButtons.forEach(button => {
      button.addEventListener('click', (e) => {
        const confirmed = confirm('Are you sure you want to delete this? This action cannot be undone.');
        if (!confirmed) {
          e.preventDefault();
          this.announce('Action cancelled');
        }
      });
    });
    
    // Add form validation before submission
    const forms = document.querySelectorAll('form');
    forms.forEach(form => {
      form.addEventListener('submit', (e) => {
        if (!this.validateForm(form)) {
          e.preventDefault();
          this.announce('Please correct the errors in the form');
        }
      });
    });
  }

  validateForm(form) {
    const requiredFields = form.querySelectorAll('[required]');
    let isValid = true;
    
    requiredFields.forEach(field => {
      if (!field.value.trim()) {
        this.showFieldError(field, 'This field is required');
        isValid = false;
      } else {
        this.clearFieldError(field);
      }
    });
    
    return isValid;
  }

  showFieldError(field, message) {
    // Remove existing error
    this.clearFieldError(field);
    
    // Add error styling
    field.classList.add('error');
    field.setAttribute('aria-invalid', 'true');
    
    // Add error message
    const errorElement = document.createElement('div');
    errorElement.className = 'field-error';
    errorElement.textContent = message;
    errorElement.setAttribute('role', 'alert');
    errorElement.id = `${field.id || field.name}-error`;
    
    field.parentNode.insertBefore(errorElement, field.nextSibling);
    field.setAttribute('aria-describedby', errorElement.id);
  }

  clearFieldError(field) {
    field.classList.remove('error');
    field.setAttribute('aria-invalid', 'false');
    
    const errorElement = field.parentNode.querySelector('.field-error');
    if (errorElement) {
      errorElement.remove();
    }
  }

  enhanceErrorMessages() {
    // Find existing error messages and enhance them
```

```typescript
  // Helper method to check if concepts are related
  areConceptsRelated(concept1: string, concept2: string): boolean {
    const relatedTerms = {
      'video': ['watch', 'play', 'media', 'content'],
      'channel': ['videos', 'creator', 'upload', 'content'],
      'profile': ['account', 'settings', 'user', 'personal'],
      'login': ['sign', 'authenticate', 'access', 'account'],
      'upload': ['add', 'create', 'post', 'share']
    };
    
    for (const [key, terms] of Object.entries(relatedTerms)) {
      if (concept1.includes(key) && terms.some(term => concept2.includes(term))) {
        return true;
      }
      if (concept2.includes(key) && terms.some(term => concept1.includes(term))) {
        return true;
      }
    }
    
    return false;
  }
```
